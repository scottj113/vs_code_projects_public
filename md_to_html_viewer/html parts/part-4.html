Sanitizer (only needed when raw HTML is enabled) ========================================================================== */ function sanitize(root) { const walker = document.createTreeWalker(root, NodeFilter.SHOW_ELEMENT); const doomed = []; let el; while ((el = walker.nextNode())) { const tag = el.tagName.toLowerCase(); if (tag === "input" && el.type === "checkbox" && el.disabled) continue; // our task-list boxes if (!OK_TAGS.has(tag)) { doomed.push(el); continue; } for (const attr of Array.from(el.attributes)) { const name = attr.name.toLowerCase(); if (name.startsWith("data-")) continue; if (!OK_ATTRS.has(name)) { el.removeAttribute(attr.name); continue; } if (URL_ATTRS.has(name)) { const clean = safeUrl(attr.value); if (clean !== attr.value) el.setAttribute(attr.name, clean); } } } // Unwrap rather than delete, so text content survives a stray
