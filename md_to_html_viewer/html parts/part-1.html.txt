<script>
/* Applied before first paint so a saved palette does not flash the default. */
try {
  var s = JSON.parse(localStorage.getItem("mdv:appearance") || "null");
  if (s && s.mode) {
    // Mirror activeMode(): "Match system" means the OS decides, so resolve it
    // here too or a system-following user gets a flash of the pinned palette.
    var m = s.auto === true
      ? (window.matchMedia("(prefers-color-scheme: dark)").matches ? "dark" : "light")
      : s.mode;
    var fam = s.family && s.family[m];
    var lvl = s.level && s.level[m];
    if (fam && lvl) document.documentElement.setAttribute("data-pal", fam + "-" + lvl);
  }
  var f = Number(localStorage.getItem("mdv:fontSize"));
  if (f) {
    document.documentElement.style.setProperty("--doc-size", f + "px");
    document.documentElement.style.setProperty("--doc-zoom", (f / 15.5).toFixed(4));
  }
  if (localStorage.getItem("mdv:bold") === "1") {
    document.documentElement.style.setProperty("--doc-weight", "600");
    document.documentElement.style.setProperty("--doc-strong", "800");
  }
} catch (e) {}
</script>
